-- 4. BYPASS & TWEEN ENGINE: Smooth movement coupled with structural Velocity Nullification
local function smoothTweenToPad(targetPart)
    local character = LocalPlayer.Character
    local hrp = character and character:FindFirstChild("HumanoidRootPart")
    
    if hrp and targetPart then
        -- Calculate distance natively to dynamic speed conversion
        local distance = (hrp.Position - targetPart.Position).Magnitude
        local dynamicSpeed = 140 -- Fast but optimal to avoid network validation kicks
        local duration = distance / dynamicSpeed
        
        -- Target spatial matrix offset (Explicitly positioned exactly 3 studs directly above the Pad)
        local targetCFrame = targetPart.CFrame * CFrame.new(0, 3, 0)
        
        local tweenInfo = TweenInfo.new(duration, Enum.EasingStyle.Linear)
        local tween = TweenService:Create(hrp, tweenInfo, {CFrame = targetCFrame})
        
        tween:Play()
        
        -- ANTI-CHEAT BYPASS: Actively monitor and isolate physics velocities during spatial translation loop
        local connection
        connection = game:GetService("RunService").Heartbeat:Connect(function()
            if tween.PlaybackState == Enum.TweenStatus.Playing then
                hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
            else
                connection:Disconnect()
            end
        end)
        
        tween.Completed:Wait()
        
        -- Extra security structural clamp upon landing inside the server synchronization phase
        hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        hrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
    end
end

-- 5. PERFORMANCE OPTIMIZATION LOOP: Heuristic local scanning routines targeting index folders directly
task.spawn(function()
    while true do
        if _G.AutoFarmActive then
            local targetWinPad = nil
            
            -- Light spatial lookup algorithm prioritizing parent directories rather than brute structural descendant scanning
            local stagesDirectory = workspace:FindFirstChild("Stages") or workspace:FindFirstChild("Worlds") or workspace:FindFirstChild("Maps")
            
            if stagesDirectory then
                -- Direct structural indexing bypassing iteration layers
                local targetStageFolder = stagesDirectory:FindFirstChild(TARGET_STAGE)
                if targetStageFolder then
                    targetWinPad = targetStageFolder:FindFirstChildWhichIsA("BasePart") 
                        or targetStageFolder:FindFirstChild("Win") 
                        or targetStageFolder:FindFirstChild("Pad") 
                        or targetStageFolder:FindFirstChild("Goal")
                end
            end
            
            -- Fallback optimization scanning strategy via shallow root array indexing
            if not targetWinPad then
                for _, obj in ipairs(workspace:GetChildren()) do
                    if obj:IsA("BasePart") and obj.Name == TARGET_STAGE then
                        targetWinPad = obj
                        break
                    elseif obj:IsA("Model") and obj.Name == TARGET_STAGE then
                        targetWinPad = obj:FindFirstChildWhichIsA("BasePart")
                        break
                    end
                end
            end
            
            -- Fallback strategy targeting Neon asset signatures across spatial layout boundaries
            if not targetWinPad then
                for _, obj in ipairs(workspace:GetChildren()) do
                    if obj:IsA("BasePart") and obj.Material == Enum.Material.Neon and obj.Size.X > 4 then
                        -- Check if object fits spatial characteristics of standard win points
                        if obj.Name:lower():find("win") or obj.Name:lower():find("pad") or obj.Name:lower():find("finish") then
                            targetWinPad = obj
                            break
                        end
                    end
                end
            end
            
            -- Execution of tween routine if target asset matrix evaluates cleanly
            if targetWinPad then
                smoothTweenToPad(targetWinPad)
                task.wait(FARM_YIELD_DELAY) -- Safe execution window processing
            else
                task.wait(1) -- Micro rest state to optimize frame budget allocation on non-indexed phases
            end
        else
            task.wait(0.5) -- Low-overhead idle processing phase when UI state evaluates to inactive
        end
    end
end)

    end
